cmake_minimum_required(VERSION 3.22.1)
project(mbloader_water_reflections)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(mbmods SHARED native/mbmods_wrapper.cpp native/water_reflections.cpp)
target_compile_definitions(mbmods PRIVATE ANDROID WATER_REFLECTIONS_VERSION="0.1.0")
find_library(log-lib log)
find_library(android-lib android)
find_library(egl-lib EGL)
find_library(gles-lib GLESv3)
find_library(dl-lib dl)
target_link_libraries(mbmods ${android-lib} ${egl-lib} ${gles-lib} ${dl-lib} ${log-lib})
